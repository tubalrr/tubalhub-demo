rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    function signedIn() { return request.auth != null; }
    function realUser() {
      return signedIn() && request.auth.token.firebase.sign_in_provider != 'anonymous';
    }
    function adminUser() {
      return realUser() && request.auth.token.admin == true;
    }
    function ownsNew() { return realUser() && request.resource.data.uid == request.auth.uid; }
    function ownsExisting() { return realUser() && resource.data.uid == request.auth.uid; }

    function chatAllowed() {
      let u = get(/databases/$(database)/documents/users/$(request.auth.uid)).data;
      return realUser()
        && (
          !('chatStatus' in u)
          || u.chatStatus == 'active'
          || (u.chatStatus == 'banned'
              && 'banUntil' in u
              && u.banUntil != null
              && u.banUntil <= request.time)
        )
        && (
          !('mutedUntil' in u)
          || u.mutedUntil == null
          || u.mutedUntil <= request.time
        );
    }

    match /globalChats/{messageId} {
      allow read: if chatAllowed();
      allow create: if chatAllowed()
        && ownsNew()
        && request.resource.data.channel == 'global-chat'
        && request.resource.data.type in ['text', 'image', 'gif']
        && (request.resource.data.type != 'text' || (request.resource.data.text is string && request.resource.data.text.size() > 0 && request.resource.data.text.size() <= 500))
        && (request.resource.data.type == 'text' || request.resource.data.mediaUrl is string);
      allow update: if (chatAllowed() && ownsExisting() && request.resource.data.uid == resource.data.uid) || adminUser();
      allow delete: if ownsExisting() || adminUser();
    }

    match /globalChatReplies/{replyId} {
      allow read: if chatAllowed();
      allow create: if chatAllowed() && ownsNew()
        && request.resource.data.messageId is string
        && request.resource.data.text is string
        && request.resource.data.text.size() > 0
        && request.resource.data.text.size() <= 500;
      allow update: if (chatAllowed() && ownsExisting() && request.resource.data.uid == request.auth.uid) || adminUser();
      allow delete: if ownsExisting() || adminUser();
    }

    match /globalChatReactions/{reactionId} {
      allow read: if chatAllowed();
      allow create: if chatAllowed() && ownsNew()
        && request.resource.data.uid == request.auth.uid
        && (request.resource.data.messageId is string || request.resource.data.replyId is string)
        && request.resource.data.reactions is list
        && request.resource.data.reactions.size() <= 10;
      allow update: if (chatAllowed() && ownsExisting() && request.resource.data.uid == request.auth.uid
        && (request.resource.data.messageId == resource.data.messageId || request.resource.data.replyId == resource.data.replyId)
        && request.resource.data.reactions is list && request.resource.data.reactions.size() <= 10) || adminUser();
      allow delete: if ownsExisting() || adminUser();
    }

    match /presence/{userId} {
      allow read: if realUser();
      allow create, update: if realUser() && userId == request.auth.uid;
      allow delete: if realUser() && userId == request.auth.uid;
    }

    match /users/{userId} {
      allow read: if realUser();
      allow create: if realUser() && userId == request.auth.uid;
      allow update: if adminUser()
        || (realUser() && userId == request.auth.uid && request.resource.data.uid == resource.data.uid
          && request.resource.data.diff(resource.data).affectedKeys().hasNone([
            'chatStatus','warnings','mutedUntil','banReason','moderatedBy','moderatedAt','banned','bannedAt','bannedBy','banUntil'
          ]));
      allow delete: if realUser() && userId == request.auth.uid;
    }

    match /globalChat/{messageId} {
      allow read, delete: if adminUser();
      allow create, update: if false;
    }

    match /adminQuotes/{quoteId} { allow read, write: if adminUser(); }

    match /messages/{messageId} {
      allow read: if realUser() && (adminUser() || request.auth.uid in resource.data.participants);
      allow create: if realUser()
        && request.resource.data.uid == request.auth.uid
        && request.resource.data.senderId == request.auth.uid
        && request.resource.data.receiverId is string
        && request.resource.data.receiverId != request.auth.uid
        && request.resource.data.participants is list
        && request.resource.data.participants.size() == 2
        && request.auth.uid in request.resource.data.participants
        && request.resource.data.receiverId in request.resource.data.participants
        && request.resource.data.text is string
        && request.resource.data.text.size() > 0
        && request.resource.data.text.size() <= 500
        && request.resource.data.type == 'text';
      allow update: if realUser() && (adminUser() || (resource.data.uid == request.auth.uid && resource.data.senderId == request.auth.uid && request.resource.data.uid == resource.data.uid && request.resource.data.senderId == resource.data.senderId && request.resource.data.receiverId == resource.data.receiverId && request.resource.data.participants == resource.data.participants));
      allow delete: if realUser() && (adminUser() || (resource.data.uid == request.auth.uid && request.resource.data.senderId == request.auth.uid));
    }

    match /videoCalls/{callId} {
      allow create: if realUser() && request.resource.data.callerId == request.auth.uid && request.resource.data.calleeId is string && request.resource.data.calleeId != request.auth.uid && request.resource.data.status == 'ringing';
      allow read: if realUser() && (resource.data.callerId == request.auth.uid || resource.data.calleeId == request.auth.uid);
      allow update: if realUser() && (resource.data.callerId == request.auth.uid || resource.data.calleeId == request.auth.uid) && request.resource.data.callerId == resource.data.callerId && request.resource.data.calleeId == resource.data.calleeId;
      allow delete: if realUser() && (resource.data.callerId == request.auth.uid || resource.data.calleeId == request.auth.uid);
      match /candidates/{candidateId} {
        allow read: if realUser() && (get(/databases/$(database)/documents/videoCalls/$(callId)).data.callerId == request.auth.uid || get(/databases/$(database)/documents/videoCalls/$(callId)).data.calleeId == request.auth.uid);
        allow create: if realUser() && request.resource.data.senderId == request.auth.uid && (get(/databases/$(database)/documents/videoCalls/$(callId)).data.callerId == request.auth.uid || get(/databases/$(database)/documents/videoCalls/$(callId)).data.calleeId == request.auth.uid);
        allow delete: if realUser() && (get(/databases/$(database)/documents/videoCalls/$(callId)).data.callerId == request.auth.uid || get(/databases/$(database)/documents/videoCalls/$(callId)).data.calleeId == request.auth.uid);
      }
    }

    match /news/{newsId} { allow read: if true; allow create, update, delete: if adminUser(); }
    match /announcements/{announcementId} { allow read: if true; allow create, update, delete: if adminUser(); }

    match /facebookVideos/{videoId} {
      allow read: if true;
      allow create: if adminUser()
        && request.resource.data.title is string && request.resource.data.title.size() <= 200
        && request.resource.data.url is string && request.resource.data.url.size() > 0 && request.resource.data.url.size() <= 2000
        && (!('thumbnail' in request.resource.data) || request.resource.data.thumbnail is string)
        && request.resource.data.createdBy == request.auth.uid;
      allow update: if adminUser()
        && request.resource.data.title is string && request.resource.data.title.size() <= 200
        && request.resource.data.url is string && request.resource.data.url.size() > 0 && request.resource.data.url.size() <= 2000;
      allow delete: if adminUser();
    }

    match /{document=**} { allow read, write: if false; }
  }
}
